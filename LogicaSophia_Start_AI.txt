#!/usr/bin/env bash
# LogicaSophia — standalone Linux AI, one-file USB launcher
# Executes explicitly when run with: bash LogicaSophia_Start_AI.txt
set -Eeuo pipefail

PROJECT_URL="https://github.com/I2PS-Engineering/LogicaSophia.git"
PROJECT="$HOME/LogicaSophia"
RUNTIME="$HOME/.local/share/logicasophia/llama.cpp"
LOGDIR="$HOME/.local/share/logicasophia/logs"
PORT=8090
# This is only a replaceable initial test model, NOT LogicaSophia's architecture.
MODEL_HF="${LOGICASOPHIA_MODEL_HF:-bartowski/SmolLM2-1.7B-Instruct-GGUF:Q4_K_M}"
mkdir -p "$LOGDIR"

if [[ "$(uname -s)" != "Linux" ]]; then echo 'Linux is required' >&2; exit 1; fi
if ! command -v python3 >/dev/null; then echo 'Python 3 is required' >&2; exit 1; fi

echo '=== LogicaSophia standalone: install + local AI agent ==='
if ! command -v git >/dev/null || ! command -v cmake >/dev/null || ! command -v g++ >/dev/null || ! command -v curl >/dev/null; then
  echo 'Installing build dependencies (Ubuntu). This may ask for your sudo password.'
  sudo apt-get update
  sudo apt-get install -y git cmake build-essential libcurl4-openssl-dev libssl-dev curl ca-certificates
fi
if [[ ! -d "$PROJECT/.git" ]]; then
  git clone "$PROJECT_URL" "$PROJECT"
else
  # Only fast-forward a clean working tree. Never reset/rewrite user changes.
  if ! git -C "$PROJECT" diff --quiet || ! git -C "$PROJECT" diff --cached --quiet; then
    echo 'Local changes found in LogicaSophia. Review them and update manually.' >&2
    exit 1
  fi
  git -C "$PROJECT" fetch origin main
  git -C "$PROJECT" merge --ff-only origin/main
fi
cd "$PROJECT"
export PYTHONPATH="$PROJECT/src"

echo '=== Checking original LogicaSophia agent tests ==='
python3 -m unittest discover -s tests -q
python3 scripts/m3_smoke.py

if [[ ! -d "$RUNTIME/.git" ]]; then
  mkdir -p "$(dirname "$RUNTIME")"
  git clone --depth 1 https://github.com/ggml-org/llama.cpp.git "$RUNTIME"
fi
if [[ ! -x "$RUNTIME/build/bin/llama-server" ]]; then
  echo 'Building local llama.cpp inference server (CPU).'
  cmake -S "$RUNTIME" -B "$RUNTIME/build" -DCMAKE_BUILD_TYPE=Release -DLLAMA_CURL=ON
  cmake --build "$RUNTIME/build" --target llama-server -j 4
fi

if curl -fsS --max-time 2 "http://127.0.0.1:$PORT/v1/models" >/dev/null 2>&1; then
  echo "A compatible local model server is already responding on port $PORT."
  STARTED_PID=''
else
  echo 'Starting replaceable local GGUF test model (download on first run).'
  "$RUNTIME/build/bin/llama-server" \
    -hf "$MODEL_HF" \
    --host 127.0.0.1 --port "$PORT" -ngl 0 -t 4 -c 4096 \
    > "$LOGDIR/model-server.log" 2>&1 &
  STARTED_PID=$!
  echo "Backend log: $LOGDIR/model-server.log"
fi
cleanup() {
  if [[ -n "${STARTED_PID:-}" ]]; then kill "$STARTED_PID" 2>/dev/null || true; fi
}
trap cleanup EXIT

printf 'Initializing local AI server'
while ! curl -fsS --max-time 2 "http://127.0.0.1:$PORT/v1/models" >/dev/null 2>&1; do
  if [[ -n "${STARTED_PID:-}" ]] && ! kill -0 "$STARTED_PID" 2>/dev/null; then
    echo; echo 'Model server exited. See the backend log:' >&2
    tail -n 35 "$LOGDIR/model-server.log" >&2
    exit 1
  fi
  printf '.'
  sleep 3
done
printf '\nReady.\n'

echo 'Workspace to inspect (press Enter for included demo):'
read -r WORKSPACE || true
WORKSPACE="${WORKSPACE:-$PROJECT/fixtures/project_fixture}"
if [[ ! -d "$WORKSPACE" ]]; then echo 'Workspace directory not found.' >&2; exit 1; fi

echo 'Enter an engineering task, or type EXIT to quit.'
echo 'Example: Explain hello.py and identify what the function returns.'
while true; do
  printf '\nLogicaSophia > '
  if ! IFS= read -r GOAL; then break; fi
  [[ "$GOAL" == "EXIT" || "$GOAL" == "exit" ]] && break
  [[ -z "$GOAL" ]] && continue
  python3 -m logicasophia agent --goal "$GOAL" --workspace "$WORKSPACE" \
    --results "$HOME/.local/share/logicasophia/results" \
    --backend-url "http://127.0.0.1:$PORT" || true
done

echo 'LogicaSophia stopped. Reports are in ~/.local/share/logicasophia/results'
